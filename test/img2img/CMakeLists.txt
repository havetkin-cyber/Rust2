cmake_minimum_required(VERSION 3.22)
project(test_img2img C CXX)
set(CMAKE_CXX_STANDARD 17)
include(FetchContent)
FetchContent_Declare(sdggml URL https://github.com/leejet/ggml/archive/eeda9ec48fc84a3df0bb9dcc19ae9053075a2539.tar.gz)
FetchContent_Populate(sdggml)
FetchContent_Declare(sdcpp URL https://github.com/leejet/stable-diffusion.cpp/archive/532d2c15c430a785a712b2d94281f04e733008a4.tar.gz)
set(SD_GGML_SOURCE_DIR "${sdggml_SOURCE_DIR}" CACHE PATH "" FORCE)
set(SD_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SD_WEBP OFF CACHE BOOL "" FORCE)
set(SD_WEBM OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE ON CACHE BOOL "" FORCE)
set(GGML_OPENMP ON CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(sdcpp)
add_executable(test_img2img test.cpp)
target_include_directories(test_img2img PRIVATE ${sdcpp_SOURCE_DIR}/include ${sdcpp_SOURCE_DIR}/thirdparty)
target_link_libraries(test_img2img PRIVATE stable-diffusion)
